cmake_minimum_required(VERSION 3.16)
project(joint_controller LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Hardware controller requires Linux; pure offline tests can be compiled separately.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(joint_controller_core
    src/jc_shm_adapter.cpp src/jc_target_adapter.cpp src/state_machine.cpp
    src/data_logger.cpp src/control_loop.cpp src/trajectory_generator.cpp
    src/output_manager.cpp src/driver_manager.cpp src/force_sensor_reader.cpp src/force_processor.cpp)
target_include_directories(joint_controller_core PUBLIC include)
target_link_libraries(joint_controller_core PUBLIC Threads::Threads rt)
target_compile_definitions(joint_controller_core PRIVATE _GNU_SOURCE)
target_compile_options(joint_controller_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
add_executable(joint_controller src/main.cpp)
target_link_libraries(joint_controller PRIVATE joint_controller_core)
target_compile_options(joint_controller PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
file(COPY tools/jointctl DESTINATION ${CMAKE_BINARY_DIR}
     FILE_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
include(CTest)
if(BUILD_TESTING)
    add_executable(offline_tests tests/offline_tests.cpp src/state_machine.cpp src/trajectory_generator.cpp)
    target_include_directories(offline_tests PRIVATE include)
    target_compile_options(offline_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    add_test(NAME offline_contracts COMMAND offline_tests)
    add_executable(experiment_tests tests/experiment_tests.cpp src/force_processor.cpp)
    target_include_directories(experiment_tests PRIVATE include)
    target_compile_options(experiment_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    add_test(NAME experiment_contracts COMMAND experiment_tests)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME yaml_contracts COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_config.py)
endif()
